# Основные команды
php minion scheduler --stats
php minion scheduler --status
php minion scheduler --list
php minion scheduler --task=example_task
php minion scheduler --reset=example_task

# Запуск планировщика
php minion scheduler
php minion scheduler --force



# Обычный запуск (может показать сообщение о блокировке)
php minion scheduler

# Принудительный запуск (игнорирует блокировку)
php minion scheduler --force

# Только снять блокировку
php minion scheduler --unlock

# Проверить статус блокировки
php minion scheduler --status


# Просмотр зарегистрированных задач
php minion taskmanager --list

# Создание новой задачи
php minion taskmanager --create --name="new_task" --class="Task_Example" --schedule="*/10 * * * *" --parameters='{"output_file":"/path/to/file.txt"}'

# Включение/выключение задачи
php minion taskmanager --enable --name="example_task"
php minion taskmanager --disable --name="example_task"

# Удаление задачи
php minion taskmanager --delete --name="old_task"


# Установка таблиц
php minion schedulerinstall --install

# Проверка статуса
php minion schedulerinstall --status

# Удаление (осторожно!)
php minion schedulerinstall --uninstall

# Запуск планировщика (проверит установку автоматически)
php minion scheduler


регистрация задач
    --list - список всех задач

    --show --name=NAME - детальная информация о задаче

    --create --name=NAME --class=CLASS --schedule=SCHEDULE - создать задачу

    --update --name=NAME - обновить задачу

    --delete --name=NAME - удалить задачу

    --enable/--disable --name=NAME - включить/выключить задачу
