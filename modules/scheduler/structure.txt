# Запуск планировщика
php minion scheduler

# Показать статус
php minion scheduler --status

# Показать детальную статистику
php minion scheduler --stats

# Сбросить статистику задачи
php minion scheduler --reset=example_task

# Запустить конкретную задачу
php minion scheduler --task=cleanup_task

# Список задач
php minion scheduler --list
